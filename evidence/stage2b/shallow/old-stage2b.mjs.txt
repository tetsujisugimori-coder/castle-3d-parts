import {readFileSync,writeFileSync,mkdirSync,existsSync,realpathSync} from 'node:fs';
import {resolve,dirname} from 'node:path';import {fileURLToPath} from 'node:url';
import {hash,digest,canonical,validateSnapshot} from './research-cad.mjs';
export const ROOT=resolve(dirname(fileURLToPath(import.meta.url)),'..');
export const BASE='34f43b3b1765de7a52994adb7175dc2c146f8db3';
const must=(x,m)=>{if(!x)throw Error(m)};const same=(a,b)=>canonical(a)===canonical(b);const read=(root,p)=>JSON.parse(readFileSync(resolve(root,p)));const key=p=>p.id+'@'+p.version;
export const RESEARCH_HASH='0454925a1805193a13efa123d025244857051a7032d34f31aff4454dde1115af';
export function preserve(root=ROOT){const b=readFileSync(resolve(root,'design/stage2b/baseline.json'));must(hash(b)==='bde65e9faa1ef3fa210e4489400de56e9c63648e2b3cac124b12d125a6245fcc','BASELINE_TAMPER');const base=JSON.parse(b);must(base.commit===BASE&&Object.keys(base.hashes).length===451,'BASELINE_SET');for(const [p,h]of Object.entries(base.hashes)){must(existsSync(resolve(root,p)),'ORIGINAL_MISSING '+p);must(hash(readFileSync(resolve(root,p)))===h,'ORIGINAL_CHANGED '+p)}return 451}
export function validateResearch(root=ROOT){const b=readFileSync(resolve(root,'design/stage2b/research-snapshot.json'));must(hash(b)===RESEARCH_HASH,'RESEARCH_PIN');const s=JSON.parse(b);must(s.records.length===27&&s.commit===BASE,'RESEARCH_SET');validateSnapshot(s,root,true);return s}
export function validateLedger(data){must(data.schema_version===1&&Array.isArray(data.revisions)&&data.revisions.length>0,'LEDGER_SCHEMA');const by=new Map;for(const r of data.revisions){must(typeof r.id==='string'&&!by.has(r.id),'REVISION_ID');by.set(r.id,r)}const seen=new Set,active=new Set;function visit(r){must(!active.has(r.id),'REVISION_CYCLE');if(seen.has(r.id))return;active.add(r.id);if(r.parent!==null){must(by.has(r.parent),'PARENT_MISSING');visit(by.get(r.parent));must(r.parent_sha256===by.get(r.parent).sha256,'PARENT_HASH')}else must(r.parent_sha256===null,'ROOT_PARENT_HASH');active.delete(r.id);seen.add(r.id)}for(const r of data.revisions)visit(r);
for(const r of data.revisions){must(r.basis==='D'&&r.cad?.eligibility==='reference_only'&&r.cad.formal_adoption===false,'FORMAL_ADOPTION');must(r.research_snapshot_sha256===RESEARCH_HASH,'RESEARCH_REFERENCE');must(same(r.decision_ids,['DEC-0001','DEC-0002']),'DECISION_REFERENCE');must(typeof r.reason==='string'&&r.reason.trim(),'REVISION_REASON');must(Array.isArray(r.parts)&&r.parts.length>0,'PARTS');const parts=new Map;for(const p of r.parts){must(typeof p.id==='string'&&typeof p.version==='string'&&!parts.has(key(p)),'PART_ID');must(['post','rail','unit'].includes(p.kind)&&p.basis==='D','PART_CLASSIFICATION');must(p.parameters&&Object.values(p.parameters).every(x=>typeof x==='number'&&Number.isFinite(x)&&x>0),'PARAMETER_TYPE');parts.set(key(p),p)}const done=new Set,stack=new Set;function walk(p){const k=key(p);must(!stack.has(k),'PART_CYCLE');if(done.has(k))return;stack.add(k);must(Array.isArray(p.depends_on),'DEPENDENCIES');for(const d of p.depends_on){must(parts.has(key(d)),'PART_REFERENCE');walk(parts.get(key(d)))}stack.delete(k);done.add(k)}for(const p of r.parts)walk(p);
for(const p of r.parts){const q=p.parameters;if(p.kind==='post')must(q.height_m>=.1&&q.height_m<=10&&q.width_m>=.01&&q.width_m<=1,'POST_RANGE');if(p.kind==='unit'){must(Number.isInteger(q.count)&&q.count>=2&&q.count<=30&&Number.isInteger(q.rails)&&q.rails>=0&&q.rails<=6,'UNIT_COUNT');const post=parts.get(key(p.depends_on.find(d=>parts.get(key(d)).kind==='post')||{}));const rail=parts.get(key(p.depends_on.find(d=>parts.get(key(d)).kind==='rail')||{}));must(post&&rail&&p.depends_on.length===2,'UNIT_COMPONENTS');must(q.spacing_m>post.parameters.width_m&&q.spacing_m<=10,'SPACING');must(Math.abs(rail.parameters.length_m-((q.count-1)*q.spacing_m+post.parameters.width_m))<1e-9,'RAIL_LENGTH');must(Math.abs(rail.parameters.width_m-post.parameters.width_m*.66)<1e-9&&Math.abs(rail.parameters.height_m-post.parameters.width_m*.56)<1e-9,'RAIL_SECTION')}}
const a=r.assembly;must(a?.version===r.id&&typeof a.id==='string'&&Array.isArray(a.instances)&&a.instances.length>=2,'ASSEMBLY');const ids=new Set;for(const i of a.instances){must(typeof i.id==='string'&&!ids.has(i.id),'INSTANCE_ID');ids.add(i.id);must(parts.has(key(i.part))&&parts.get(key(i.part)).kind==='unit','INSTANCE_REFERENCE');for(const f of ['position_m','rotation_deg'])must(Array.isArray(i[f])&&i[f].length===3&&i[f].every(x=>typeof x==='number'&&Number.isFinite(x)),'PLACEMENT');must(i.rotation_deg.every(x=>x===0),'UNSUPPORTED_ROTATION')}
const {sha256,...body}=r;must(sha256===digest(body),'REVISION_HASH');if(r.parent){const parent=by.get(r.parent);for(const p of parent.parts){const retained=parts.get(key(p));must(retained&&same(retained,p),'PAST_PART_CHANGED')}}}
return true}
export function geometry(r){const parts=new Map(r.parts.map(p=>[key(p),p]));const units=r.assembly.instances.map(i=>{const u=parts.get(key(i.part)),p=parts.get(key(u.depends_on.find(d=>parts.get(key(d)).kind==='post')));const width=(u.parameters.count-1)*u.parameters.spacing_m+p.parameters.width_m;return {id:i.id,unit_revision:i.part.version,position_m:i.position_m,dimensions_m:[width,p.parameters.width_m*1.66,p.parameters.height_m],posts:u.parameters.count,rails:u.parameters.rails,spacing_m:u.parameters.spacing_m,post_width_m:p.parameters.width_m}});const lo=[Infinity,Infinity,Infinity],hi=[-Infinity,-Infinity,-Infinity];for(const u of units){const [w,d,h]=u.dimensions_m,[x,y,z]=u.position_m;const mins=[x-w/2,y-u.post_width_m*1.16,z],maxs=[x+w/2,y+u.post_width_m/2,z+h];for(let n=0;n<3;n++){lo[n]=Math.min(lo[n],mins[n]);hi[n]=Math.max(hi[n],maxs[n])}}return {units,dimensions_m:hi.map((x,i)=>x-lo[i]),solid_count:units.reduce((n,u)=>n+u.posts+u.rails,0)}}
export function diff(a,b){const ga=geometry(a),gb=geometry(b);return {from:a.id,to:b.id,from_sha256:a.sha256,to_sha256:b.sha256,unit:'m',parameters_and_parts:{changed:!same(a.parts,b.parts),before:a.parts,after:b.parts},placements:{changed:!same(a.assembly.instances,b.assembly.instances),before:a.assembly.instances,after:b.assembly.instances},geometry:{before:ga,after:gb,dimensions_delta_m:gb.dimensions_m.map((x,i)=>x-ga.dimensions_m[i])},research:{changed:a.research_snapshot_sha256!==b.research_snapshot_sha256,before:a.research_snapshot_sha256,after:b.research_snapshot_sha256},classification:'D; reference_only; not historical measurement'}}
export function writeNew(path,data){const p=resolve(path);must(!existsSync(p),'OUTPUT_EXISTS');let ancestor=dirname(p);while(!existsSync(ancestor))ancestor=dirname(ancestor);must(realpathSync(ancestor)===ancestor,'OUTPUT_SYMLINK');mkdirSync(dirname(p),{recursive:true});writeFileSync(p,JSON.stringify(data,null,2)+'\n',{flag:'wx'})}
export function check(root=ROOT){const data=read(root,'design/stage2b/revisions.json');validateLedger(data);const research=validateResearch(root);return {status:'PASS',preserved_files:preserve(root),research_records:research.records.length,revisions:data.revisions.map(r=>({id:r.id,...geometry(r)}))}}

export function validateSpec(spec,ledger){validateLedger(ledger);must(spec.schema_version===1&&spec.basis==='D'&&spec.cad_eligibility==='reference_only','SPEC_CLASSIFICATION');must(Array.isArray(spec.revisions)&&spec.revisions.length===ledger.revisions.length,'SPEC_REVISIONS');for(const [n,r]of ledger.revisions.entries()){const s=spec.revisions[n];must(s.id===r.id&&s.parent===r.parent,'SPEC_PARENT');const parts=new Map(r.parts.map(p=>[key(p),p]));must(s.units.length===r.assembly.instances.length,'SPEC_UNITS');for(const [j,i]of r.assembly.instances.entries()){const u=parts.get(key(i.part)),p=parts.get(key(u.depends_on.find(d=>parts.get(key(d)).kind==='post'))),v=s.units[j];must(v.instance_id===i.id&&v.unit_revision===i.part.version&&same(v.position_m,i.position_m),'SPEC_PLACEMENT');must(same(v.parameters,{height:p.parameters.height_m,spacing:u.parameters.spacing_m,width:p.parameters.width_m,count:u.parameters.count,rails:u.parameters.rails}),'SPEC_PARAMETERS')}}return true}
export function expectedComponents(s){const out=[];for(const u of s.units){const p=u.parameters,[x,y,z]=u.position_m,w=p.width,L=(p.count-1)*p.spacing+w;for(let j=0;j<p.count;j++){const a=x+(j-(p.count-1)/2)*p.spacing-w/2;out.push({id:u.instance_id+'_Post_'+String(j+1).padStart(3,'0'),part_id:'PAL-POST',revision:u.unit_revision,bounds_m:[a,y-w/2,z,a+w,y+w/2,z+p.height]})}for(let j=0;j<p.rails;j++){const a=z+p.height*(j+1)/(p.rails+1)-w*.28;out.push({id:u.instance_id+'_Rail_'+String(j+1).padStart(3,'0'),part_id:'PAL-RAIL',revision:u.unit_revision,bounds_m:[x-L/2,y-w/2-w*.66,a,x+L/2,y-w/2,a+w*.56]})}}return out}
function near(a,b,tol=1e-5){return Array.isArray(a)&&a.length===b.length&&a.every((x,i)=>typeof x==='number'&&Number.isFinite(x)&&Math.abs(x-b[i])<tol)}
export function validateAudit(cad,blender,r,s){const expected=expectedComponents(s),g=geometry(r);for(const [label,a]of [['cad_created',cad.created],['cad_reopened',cad.fcstd_reopened],['blender_created',blender.created],['blend_reopened',blender.blend_reopened],['glb_reopened',blender.glb_reopened]]){must(a&&near(a.dimensions_m,g.dimensions_m),'AUDIT_DIMENSIONS '+label);must(Array.isArray(a.components)&&a.components.length===expected.length,'AUDIT_COMPONENTS '+label);const by=new Map(a.components.map(c=>[c.id,c]));must(by.size===expected.length,'AUDIT_COMPONENT_IDS');for(const c of expected){const v=by.get(c.id);must(v&&v.part_id===c.part_id&&v.revision===c.revision&&near(v.bounds_m,c.bounds_m),'AUDIT_PLACEMENT '+label+' '+c.id)}}must(cad.created.solids===g.solid_count&&cad.fcstd_reopened.solids===g.solid_count&&cad.step_reopened===true&&cad.step_components===g.solid_count&&cad.edit_recompute_revert===true,'CAD_RELOAD');for(const a of [cad.created,cad.fcstd_reopened])must(a.intersection_checks===expected.length*(expected.length-1)/2&&typeof a.max_overlap_mm3==='number'&&a.max_overlap_mm3>=0&&a.max_overlap_mm3<1e-5,'INTERFERENCE');must(same(cad.techdraw?.views,['FrontView','TopView','SideView'])&&near(cad.techdraw.dimension_bounds_m,g.dimensions_m),'DRAWING_REFERENCE');must(cad.techdraw.native_linked_dimensions==='NOT_TESTED: headless projected vertex extraction returns no vertices; invalid zero dimensions removed'&&(!Object.hasOwn(cad.techdraw,'native_dimension_status')||cad.techdraw.native_dimension_status==='NOT_TESTED')&&same(cad.techdraw.native_dimensions,[]),'NATIVE_DIMENSION_STATUS');must(cad.techdraw.native_scale===.02,'NATIVE_VIEW_SCALE');return true}
export const REQUIRED_OUTPUTS=['assembly.FCStd','assembly.step','unit.FCStd','unit.step','assembly.blend','assembly.glb','mesh-transfer.json','cad-audit.json','blender-audit.json','drawing.svg','drawing-measurements.json'];
export function validateManifest(m,revision,specHash,readBytes){must(m.schema_version===1&&m.revision===revision&&m.basis==='D'&&m.cad_eligibility==='reference_only','MANIFEST_ID');must(m.spec_sha256===specHash,'MANIFEST_SPEC');const scripts=['cad/stage2b/build.py','blender/stage2b/build.py','cad/stage2b/run.py','cad/stage2b/drawing.py'];must(same(Object.keys(m.source_scripts??{}).sort(),scripts.sort()),'MANIFEST_SCRIPTS');for(const p of scripts)must(m.source_scripts[p]===hash(readBytes('script',p)),'MANIFEST_SOURCE_HASH');must(m.outputs&&REQUIRED_OUTPUTS.every(p=>Object.hasOwn(m.outputs,p)),'MANIFEST_REQUIRED');for(const [p,h]of Object.entries(m.outputs)){must(/^[\w.-]+$/.test(p)&&/^[a-f0-9]{64}$/.test(h),'MANIFEST_PATH');must(hash(readBytes('output',p))===h,'MANIFEST_OUTPUT_HASH '+p)}return true}

export function validateDrawingMeasurements(m,r,s){const g=geometry(r);const expected={'overall-width':g.dimensions_m[0]*1000,'overall-height':g.dimensions_m[2]*1000,'side-depth':g.dimensions_m[1]*1000};for(const u of s.units)expected[u.instance_id+'-post-centres']=u.parameters.spacing*1000;for(let i=1;i<=s.units[0].parameters.rails;i++)expected[s.units[0].instance_id+'-rail-centre-'+i]=s.units[0].parameters.height*1000*i/(s.units[0].parameters.rails+1);must(m.revision===r.id&&m.scale===.02&&m.unit==='mm','DRAWING_METADATA');must(Array.isArray(m.dimensions)&&m.dimensions.length===Object.keys(expected).length,'DRAWING_DIMENSION_COUNT');const ids=new Set;for(const d of m.dimensions){must(!ids.has(d.id)&&Object.hasOwn(expected,d.id),'DRAWING_DIMENSION_ID');ids.add(d.id);must(typeof d.value_mm==='number'&&Math.abs(d.value_mm-expected[d.id])<1e-6,'DRAWING_DIMENSION_VALUE');must(d.scale===.02&&Array.isArray(d.endpoints_page_mm)&&d.endpoints_page_mm.length===2&&d.endpoints_page_mm.every(p=>Array.isArray(p)&&p.length===2&&p.every(Number.isFinite)),'DRAWING_ENDPOINT');const[a,b]=d.endpoints_page_mm;const span=Math.hypot(b[0]-a[0],b[1]-a[1]);must(Math.abs(span/.02-d.value_mm)<1e-5,'DRAWING_WITNESS_SCALE')}return true}
