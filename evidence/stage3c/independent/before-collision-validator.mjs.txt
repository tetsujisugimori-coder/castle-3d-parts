import{validateMechanism as baseValidate,validateProjectionMesh as baseMesh,posePoint}from'../stage3b/contract.mjs';
export{verifyHash,angleWithin,posePoint,advance,sampleStatus,projectedTriangles}from'../stage3b/contract.mjs';
const ok=(v,m)=>{if(!v)throw Error(m)};
export function validateLock(m){const b=m.bolt;ok(b&&typeof b.id==='string'&&m.parts.some(p=>p.id===b.id)&&m.moving_ids.includes(b.id),'bolt identity');ok(b.parent==='BoltSlide','bolt parent');ok(Array.isArray(b.travel_m)&&b.travel_m.length===2&&b.travel_m.every(Number.isFinite)&&b.travel_m[0]===0&&b.travel_m[1]>0&&b.travel_m[1]<2,'bolt range');ok(b.unlock_m===0&&b.lock_m===b.travel_m[1],'bolt endpoints');ok(Array.isArray(b.axis)&&b.axis.length===3&&b.axis.every((v,i)=>v===[1,0,0][i]),'bolt CAD axis');ok(b.closure_tolerance_deg===0,'closure contract');m.lock={bolt_id:b.id,parent:b.parent,range_m:b.travel_m,axis:b.axis,unlocked_m:b.unlock_m,locked_m:b.lock_m,closed_angle_tolerance_deg:b.closure_tolerance_deg};return m;}
export function validateMechanism(m){return validateLock(baseValidate(m))}
export function transition(m,s,a){validateLock(m);ok(s&&Number.isFinite(s.angle)&&Number.isFinite(s.bolt),'状態を確定できません');ok(s.angle>=m.angle_range_deg[0]&&s.angle<=m.angle_range_deg[1]&&s.bolt>=0&&s.bolt<=m.bolt.lock_m,'状態が可動範囲外です');ok(!(s.angle!==0&&s.bolt!==0),'矛盾した機構状態です');ok(a&&['angle','bolt'].includes(a.type)&&Number.isFinite(a.value),'不正な操作です');const n={...s};if(a.type==='angle'){ok(a.value>=0&&a.value<=m.angle_range_deg[1],'角度が可動範囲外です');ok(s.bolt===0||a.value===0,'閂を完全に外してから開いてください');n.angle=a.value;}else{ok(a.value>=0&&a.value<=m.bolt.lock_m,'閂が可動範囲外です');ok(s.angle===0||a.value===0,'扉を閉鎖位置へ戻してから施錠してください');n.bolt=a.value;}return n;}
export function validateProjectionMesh(m,d){baseMesh(m,d);for(const o of d.objects)ok(o.movement===(o.id===m.bolt.id?'bolt':m.moving_ids.includes(o.id)?'door':'fixed'),'projection motion kind');return d}
export function lockSample(m,c,s){return [...c.samples,...(c.bolt_samples||[])].find(r=>Math.abs(r.angle_degrees-s.angle)<1e-8&&Math.abs((r.bolt_m??0)-s.bolt)<1e-9)}
export function validateCollision(m,c){ok(m.collision?.path==='collision.json'&&/^[a-f0-9]{64}$/.test(m.collision.sha256),'collision hash/path');ok(Number.isFinite(c.tolerance_mm3)&&c.tolerance_mm3>=0&&c.tolerance_mm3<=.001,'collision tolerance');ok(Array.isArray(c.samples)&&c.samples.length>0,'collision samples');const all=[...c.samples,...(c.bolt_samples||[])];for(const r of all){ok(Number.isFinite(r.angle_degrees)&&Number.isFinite(r.bolt_m??0),'pose quantities');transition(m,{angle:0,bolt:0},{type:'angle',value:r.angle_degrees});ok((r.bolt_m??0)>=0&&(r.bolt_m??0)<=m.bolt.lock_m&&(!(r.bolt_m>0)||r.angle_degrees===0),'collision pose');ok(Array.isArray(r.pairs)&&r.pairs.length>0,'collision pair coverage');const seen=new Set();for(const p of r.pairs){const key=[p.a,p.b].sort().join('|');ok(typeof p.a==='string'&&typeof p.b==='string'&&p.a!==p.b&&!seen.has(key),'collision identity');seen.add(key);ok(m.collision_ids.includes(p.a)&&m.collision_ids.includes(p.b),'collision unknown ID');ok(Number.isFinite(p.distance_m)&&p.distance_m>=0&&Number.isFinite(p.overlap_mm3)&&p.overlap_mm3>=0,'collision quantities');const state=p.overlap_mm3>c.tolerance_mm3?'PENETRATION':p.distance_m<1e-8?'TOUCHING':'CLEAR';ok(state===p.contact,'collision classification')}}for(const a of m.sampled_angles_deg)ok(all.some(r=>r.angle_degrees===a&&(r.bolt_m??0)===0),'door sample coverage');ok(all.some(r=>r.angle_degrees===0&&r.bolt_m===m.bolt.lock_m),'locked sample coverage');return c;}
